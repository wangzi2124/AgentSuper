```js
/**
 * 七种标准方块（Tetromino）定义与旋转逻辑
 * 使用矩阵表示形状，旋转即矩阵转置 + 反转
 */
const SHAPES = {
  I: [[1, 1, 1, 1]],
  O: [
    [1, 1],
    [1, 1],
  ],
  T: [
    [0, 1, 0],
    [1, 1, 1],
  ],
  S: [
    [0, 1, 1],
    [1, 1, 0],
  ],
  Z: [
    [1, 1, 0],
    [0, 1, 1],
  ],
  J: [
    [1, 0, 0],
    [1, 1, 1],
  ],
  L: [
    [0, 0, 1],
    [1, 1, 1],
  ],
};

/** 每种方块对应的颜色 */
const COLORS = {
  I: "#4fd1c5",
  O: "#f6c445",
  T: "#b06cf0",
  S: "#5ad469",
  Z: "#ef5b6b",
  J: "#4f8ef7",
  L: "#f0913f",
};

class Tetromino {
  /**
   * @param {string} type 方块类型（I/O/T/S/Z/J/L）
   */
  constructor(type) {
    this.type = type;
    // 深拷贝形状，避免旋转时污染原始定义
    this.matrix = SHAPES[type].map((row) => [...row]);
    this.color = COLORS[type];
    // 初始位置：水平居中，顶部
    this.x = Math.floor((CONFIG.COLS - this.matrix[0].length) / 2);
    this.y = 0;
  }

  /** 顺时针旋转 90°，返回新矩阵（不修改自身） */
  rotatedMatrix() {
    const rows = this.matrix.length;
    const cols = this.matrix[0].length;
    const result = Array.from({ length: cols }, () => Array(rows).fill(0));
    for (let r = 0; r < rows; r++) {
      for (let c = 0; c < cols; c++) {
        result[c][rows - 1 - r] = this.matrix[r][c];
      }
    }
    return result;
  }

  /** 执行旋转 */
  rotate() {
    this.matrix = this.rotatedMatrix();
  }

  /** 返回方块占据的绝对坐标列表 [{x, y}] */
  cells(matrix = this.matrix, offsetX = this.x, offsetY = this.y) {
    const list = [];
    matrix.forEach((row, r) => {
      row.forEach((val, c) => {
        if (val) list.push({ x: offsetX + c, y: offsetY + r });
      });
    });
    return list;
  }
}

/** 随机生成一个方块类型 */
function randomType() {
  const types = Object.keys(SHAPES);
  return types[Math.floor(Math.random() * types.length)];
}
```

```js
/**
 * 棋盘：负责网格数据、碰撞检测、锁定方块、消行
 */
class Board {
  constructor() {
    this.reset();
  }

  reset() {
    // 二维数组，null 表示空格，否则存储颜色
    this.grid = Array.from({ length: CONFIG.ROWS }, () =>
      Array(CONFIG.COLS).fill(null)
    );
  }

  /** 判断一组坐标是否越界或与已锁定方块重叠 */
  isCollision(cells) {
    return cells.some(({ x, y }) => {
      if (x < 0 || x >= CONFIG.COLS || y >= CONFIG.ROWS) return true;
      if (y < 0) return false; // 允许在顶部之上（生成时）
      return this.grid[y][x] !== null;
    });
  }

  /** 将方块锁定进网格 */
  lock(tetromino) {
    tetromino.cells().forEach(({ x, y }) => {
      if (y >= 0 && y < CONFIG.ROWS && x >= 0 && x < CONFIG.COLS) {
        this.grid[y][x] = tetromino.color;
      }
    });
  }

  /**
   * 消除已填满的行
   * @returns {number} 消除的行数
   */
  clearLines() {
    let cleared = 0;
    for (let r = CONFIG.ROWS - 1; r >= 0; r--) {
      if (this.grid[r].every((cell) => cell !== null)) {
        this.grid.splice(r, 1);
        this.grid.unshift(Array(CONFIG.COLS).fill(null));
        cleared++;
        r++; // 重新检查当前行（因为上方行下移了）
      }
    }
    return cleared;
  }

  /** 绘制棋盘与已锁定方块 */
  draw(ctx) {
    const { BLOCK_SIZE, COLS, ROWS } = CONFIG;
    ctx.clearRect(0, 0, COLS * BLOCK_SIZE, ROWS * BLOCK_SIZE);

    // 网格线
    ctx.strokeStyle = "rgba(255,255,255,0.05)";
    ctx.lineWidth = 1;
    for (let c = 0; c <= COLS; c++) {
      ctx.beginPath();
      ctx.moveTo(c * BLOCK_SIZE, 0);
      ctx.lineTo(c * BLOCK_SIZE, ROWS * BLOCK_SIZE);
      ctx.stroke();
    }
    for (let r = 0; r <= ROWS; r++) {
      ctx.beginPath();
      ctx.moveTo(0, r * BLOCK_SIZE);
      ctx.lineTo(COLS * BLOCK_SIZE, r * BLOCK_SIZE);
      ctx.stroke();
    }

    // 已锁定方块
    this.grid.forEach((row, r) => {
      row.forEach((color, c) => {
        if (color) drawBlock(ctx, c, r, color);
      });
    });
  }
}

/** 通用方块绘制函数（带高光效果） */
function drawBlock(ctx, col, row, color) {
  const { BLOCK_SIZE } = CONFIG;
  const x = col * BLOCK_SIZE;
  const y = row * BLOCK_SIZE;
  const pad = 1;

  ctx.fillStyle = color;
  ctx.fillRect(x + pad, y + pad, BLOCK_SIZE - pad * 2, BLOCK_SIZE - pad * 2);

  // 顶部/左侧高光
  ctx.fillStyle = "rgba(255,255,255,0.25)";
  ctx.fillRect(x + pad, y + pad, BLOCK_SIZE - pad * 2, 4);
  ctx.fillRect(x + pad, y + pad, 4, BLOCK_SIZE - pad * 2);
}
```

```js
/**
 * 计分系统：负责分数、消行、等级、最高分
 */
class ScoreSystem {
  constructor() {
    this.score = 0;
    this.lines = 0;
    this.level = 1;
    this.highScore = this.loadHighScore();
  }

  reset() {
    this.score = 0;
    this.lines = 0;
    this.level = 1;
  }

  /**
   * 根据消除行数计算并累加分数
   * 分数 = 基础分 × 当前等级
   * @param {number} clearedLines 本次消除的行数
   */
  addLines(clearedLines) {
    if (clearedLines <= 0) return;
    const base = CONFIG.LINE_SCORES[clearedLines] || 0;
    this.score += base * this.level;
    this.lines += clearedLines;
    this.updateLevel();
    this.updateHighScore();
  }

  /** 软降/硬降奖励分 */
  addDropBonus(cells, isHard) {
    const bonus = isHard ? CONFIG.HARD_DROP_BONUS : CONFIG.SOFT_DROP_BONUS;
    this.score += cells * bonus;
    this.updateHighScore();
  }

  /** 每消除 LINES_PER_LEVEL 行升一级 */
  updateLevel() {
    this.level = Math.floor(this.lines / CONFIG.LINES_PER_LEVEL) + 1;
  }

  /** 当前等级对应的下落间隔（毫秒） */
  getDropInterval() {
    const interval =
      CONFIG.BASE_DROP_INTERVAL - (this.level - 1) * CONFIG.SPEED_STEP;
    return Math.max(interval, CONFIG.MIN_DROP_INTERVAL);
  }

  updateHighScore() {
    if (this.score > this.highScore) {
      this.highScore = this.score;
      this.saveHighScore();
    }
  }

  loadHighScore() {
    try {
      return Number(localStorage.getItem(CONFIG.STORAGE_KEY)) || 0;
    } catch {
      return 0;
    }
  }

  saveHighScore() {
    try {
      localStorage.setItem(CONFIG.STORAGE_KEY, String(this.highScore));
    } catch {
      /* 忽略存储异常 */
    }
  }
}
```
