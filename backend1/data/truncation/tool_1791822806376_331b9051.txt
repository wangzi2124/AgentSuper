273: ```js
274: /**
275:  * 七种标准方块（Tetromino）定义与旋转逻辑
276:  * 使用矩阵表示形状，旋转即矩阵转置 + 反转
277:  */
278: const SHAPES = {
279:   I: [[1, 1, 1, 1]],
280:   O: [
281:     [1, 1],
282:     [1, 1],
283:   ],
284:   T: [
285:     [0, 1, 0],
286:     [1, 1, 1],
287:   ],
288:   S: [
289:     [0, 1, 1],
290:     [1, 1, 0],
291:   ],
292:   Z: [
293:     [1, 1, 0],
294:     [0, 1, 1],
295:   ],
296:   J: [
297:     [1, 0, 0],
298:     [1, 1, 1],
299:   ],
300:   L: [
301:     [0, 0, 1],
302:     [1, 1, 1],
303:   ],
304: };
305: 
306: /** 每种方块对应的颜色 */
307: const COLORS = {
308:   I: "#4fd1c5",
309:   O: "#f6c445",
310:   T: "#b06cf0",
311:   S: "#5ad469",
312:   Z: "#ef5b6b",
313:   J: "#4f8ef7",
314:   L: "#f0913f",
315: };
316: 
317: class Tetromino {
318:   /**
319:    * @param {string} type 方块类型（I/O/T/S/Z/J/L）
320:    */
321:   constructor(type) {
322:     this.type = type;
323:     // 深拷贝形状，避免旋转时污染原始定义
324:     this.matrix = SHAPES[type].map((row) => [...row]);
325:     this.color = COLORS[type];
326:     // 初始位置：水平居中，顶部
327:     this.x = Math.floor((CONFIG.COLS - this.matrix[0].length) / 2);
328:     this.y = 0;
329:   }
330: 
331:   /** 顺时针旋转 90°，返回新矩阵（不修改自身） */
332:   rotatedMatrix() {
333:     const rows = this.matrix.length;
334:     const cols = this.matrix[0].length;
335:     const result = Array.from({ length: cols }, () => Array(rows).fill(0));
336:     for (let r = 0; r < rows; r++) {
337:       for (let c = 0; c < cols; c++) {
338:         result[c][rows - 1 - r] = this.matrix[r][c];
339:       }
340:     }
341:     return result;
342:   }
343: 
344:   /** 执行旋转 */
345:   rotate() {
346:     this.matrix = this.rotatedMatrix();
347:   }
348: 
349:   /** 返回方块占据的绝对坐标列表 [{x, y}] */
350:   cells(matrix = this.matrix, offsetX = this.x, offsetY = this.y) {
351:     const list = [];
352:     matrix.forEach((row, r) => {
353:       row.forEach((val, c) => {
354:         if (val) list.push({ x: offsetX + c, y: offsetY + r });
355:       });
356:     });
357:     return list;
358:   }
359: }
360: 
361: /** 随机生成一个方块类型 */
362: function randomType() {
363:   const types = Object.keys(SHAPES);
364:   return types[Math.floor(Math.random() * types.length)];
365: }

369: 
370: ```js
371: /**
372:  * 棋盘：负责网格数据、碰撞检测、锁定方块、消行
373:  */
374: class Board {
375:   constructor() {
376:     this.reset();
377:   }
378: 
379:   reset() {
380:     // 二维数组，null 表示空格，否则存储颜色
381:     this.grid = Array.from({ length: CONFIG.ROWS }, () =>
382:       Array(CONFIG.COLS).fill(null)
383:     );
384:   }
385: 
386:   /** 判断一组坐标是否越界或与已锁定方块重叠 */
387:   isCollision(cells) {
388:     return cells.some(({ x, y }) => {
389:       if (x < 0 || x >= CONFIG.COLS || y >= CONFIG.ROWS) return true;
390:       if (y < 0) return false; // 允许在顶部之上（生成时）
391:       return this.grid[y][x] !== null;
392:     });
393:   }
394: 
395:   /** 将方块锁定进网格 */
396:   lock(tetromino) {
397:     tetromino.cells().forEach(({ x, y }) => {
398:       if (y >= 0 && y < CONFIG.ROWS && x >= 0 && x < CONFIG.COLS) {
399:         this.grid[y][x] = tetromino.color;
400:       }
401:     });
402:   }
403: 
404:   /**
405:    * 消除已填满的行
406:    * @returns {number} 消除的行数
407:    */
408:   clearLines() {
409:     let cleared = 0;
410:     for (let r = CONFIG.ROWS - 1; r >= 0; r--) {
411:       if (this.grid[r].every((cell) => cell !== null)) {
412:         this.grid.splice(r, 1);
413:         this.grid.unshift(Array(CONFIG.COLS).fill(null));
414:         cleared++;
415:         r++; // 重新检查当前行（因为上方行下移了）
416:       }
417:     }
418:     return cleared;
419:   }
420: 
421:   /** 绘制棋盘与已锁定方块 */
422:   draw(ctx) {
423:     const { BLOCK_SIZE, COLS, ROWS } = CONFIG;
424:     ctx.clearRect(0, 0, COLS * BLOCK_SIZE, ROWS * BLOCK_SIZE);
425: 
426:     // 网格线
427:     ctx.strokeStyle = "rgba(255,255,255,0.05)";
428:     ctx.lineWidth = 1;
429:     for (let c = 0; c <= COLS; c++) {
430:       ctx.beginPath();
431:       ctx.moveTo(c * BLOCK_SIZE, 0);
432:       ctx.lineTo(c * BLOCK_SIZE, ROWS * BLOCK_SIZE);
433:       ctx.stroke();
434:     }
435:     for (let r = 0; r <= ROWS; r++) {
436:       ctx.beginPath();
437:       ctx.moveTo(0, r * BLOCK_SIZE);
438:       ctx.lineTo(COLS * BLOCK_SIZE, r * BLOCK_SIZE);
439:       ctx.stroke();
440:     }
441: 
442:     // 已锁定方块
443:     this.grid.forEach((row, r) => {
444:       row.forEach((color, c) => {
445:         if (color) drawBlock(ctx, c, r, color);
446:       });
447:     });
448:   }
449: }
450: 
451: /** 通用方块绘制函数（带高光效果） */
452: function drawBlock(ctx, col, row, color) {
453:   const { BLOCK_SIZE } = CONFIG;
454:   const x = col * BLOCK_SIZE;
455:   const y = row * BLOCK_SIZE;
456:   const pad = 1;
457: 
458:   ctx.fillStyle = color;
459:   ctx.fillRect(x + pad, y + pad, BLOCK_SIZE - pad * 2, BLOCK_SIZE - pad * 2);
460: 
461:   // 顶部/左侧高光
462:   ctx.fillStyle = "rgba(255,255,255,0.25)";
463:   ctx.fillRect(x + pad, y + pad, BLOCK_SIZE - pad * 2, 4);
464:   ctx.fillRect(x + pad, y + pad, 4, BLOCK_SIZE - pad * 2);
465: }

471: /**
472:  * 计分系统：负责分数、消行、等级、最高分
473:  */
474: class ScoreSystem {
475:   constructor() {
476:     this.score = 0;
477:     this.lines = 0;
478:     this.level = 1;
479:     this.highScore = this.loadHighScore();
480:   }
481: 
482:   reset() {
483:     this.score = 0;
484:     this.lines = 0;
485:     this.level = 1;
486:   }
487: 
488:   /**
489:    * 根据消除行数计算并累加分数
490:    * 分数 = 基础分 × 当前等级
491:    * @param {number} clearedLines 本次消除的行数
492:    */
493:   addLines(clearedLines) {
494:     if (clearedLines <= 0) return;
495:     const base = CONFIG.LINE_SCORES[clearedLines] || 0;
496:     this.score += base * this.level;
497:     this.lines += clearedLines;
498:     this.updateLevel();
499:     this.updateHighScore();
500:   }
501: 
502:   /** 软降/硬降奖励分 */
503:   addDropBonus(cells, isHard) {
504:     const bonus = isHard ? CONFIG.HARD_DROP_BONUS : CONFIG.SOFT_DROP_BONUS;
505:     this.score += cells * bonus;
506:     this.updateHighScore();
507:   }
508: 
509:   /** 每消除 LINES_PER_LEVEL 行升一级 */
510:   updateLevel() {
511:     this.level = Math.floor(this.lines / CONFIG.LINES_PER_LEVEL) + 1;
512:   }
513: 
514:   /** 当前等级对应的下落间隔（毫秒） */
515:   getDropInterval() {
516:     const interval =
517:       CONFIG.BASE_DROP_INTERVAL - (this.level - 1) * CONFIG.SPEED_STEP;
518:     return Math.max(interval, CONFIG.MIN_DROP_INTERVAL);
519:   }
520: 
521:   updateHighScore() {
522:     if (this.score > this.highScore) {
523:       this.highScore = this.score;
524:       this.saveHighScore();
525:     }
526:   }
527: 
528:   loadHighScore() {
529:     try {
530:       return Number(localStorage.getItem(CONFIG.STORAGE_KEY)) || 0;
531:     } catch {
532:       return 0;
533:     }
534:   }
535: 
536:   saveHighScore() {
537:     try {
538:       localStorage.setItem(CONFIG.STORAGE_KEY, String(this.highScore));
539:     } catch {
540:       /* 忽略存储异常 */
541:     }
542:   }
543: }
