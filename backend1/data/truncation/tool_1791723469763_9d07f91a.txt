## 3. `game.js`

```javascript
(() => {
  "use strict";

  // ===== 常量配置 =====
  const COLS = 10;
  const ROWS = 20;
  const BLOCK = 30; // 每个格子像素

  const canvas = document.getElementById("board");
  const ctx = canvas.getContext("2d");
  const nextCanvas = document.getElementById("next");
  const nctx = nextCanvas.getContext("2d");

  const scoreEl = document.getElementById("score");
  const levelEl = document.getElementById("level");
  const linesEl = document.getElementById("lines");

  const overlay = document.getElementById("overlay");
  const overlayTitle = document.getElementById("overlay-title");
  const overlayText = document.getElementById("overlay-text");
  const overlayBtn = document.getElementById("overlay-btn");
  const restartBtn = document.getElementById("restart");

  // 7 种方块形状（用矩阵表示）
  const SHAPES = {
    I: [[1, 1, 1, 1]],
    O: [[1, 1], [1, 1]],
    T: [[0, 1, 0], [1, 1, 1]],
    S: [[0, 1, 1], [1, 1, 0]],
    Z: [[1, 1, 0], [0, 1, 1]],
    J: [[1, 0, 0], [1, 1, 1]],
    L: [[0, 0, 1], [1, 1, 1]],
  };

  const COLORS = {
    I: "#6ee7ff",
    O: "#ffd166",
    T: "#c084fc",
    S: "#4ade80",
    Z: "#ff6b6b",
    J: "#60a5fa",
    L: "#fb923c",
  };

  const KEYS = Object.keys(SHAPES);

  // ===== 游戏状态 =====
  let board;        // 二维数组，0 表示空，否则存颜色
  let current;      // 当前方块 { shape, color, x, y }
  let nextPiece;    // 下一个方块
  let score, level, lines;
  let dropInterval; // 下落间隔(ms)
  let dropCounter;  // 累计时间
  let lastTime;
  let paused = false;
  let gameOver = false;
  let rafId = null;

  // ===== 工具函数 =====
  function createBoard() {
    return Array.from({ length: ROWS }, () => Array(COLS).fill(0));
  }

  function randomPiece() {
    const key = KEYS[(Math.random() * KEYS.length) | 0];
    const shape = SHAPES[key].map((row) => row.slice());
    return {
      shape,
      color: COLORS[key],
      x: ((COLS - shape[0].length) / 2) | 0,
      y: 0,
    };
  }

  // 旋转矩阵（顺时针 90°）
  function rotateMatrix(matrix) {
    const rows = matrix.length;
    const cols = matrix[0].length;
    const result = Array.from({ length: cols }, () => Array(rows).fill(0));
    for (let r = 0; r < rows; r++) {
      for (let c = 0; c < cols; c++) {
        result[c][rows - 1 - r] = matrix[r][c];
      }
    }
    return result;
  }

  // 碰撞检测
  function collide(shape, offsetX, offsetY) {
    for (let r = 0; r < shape.length; r++) {
      for (let c = 0; c < shape[r].length; c++) {
        if (!shape[r][c]) continue;
        const x = offsetX + c;
        const y = offsetY + r;
        if (x < 0 || x >= COLS || y >= ROWS) return true;
        if (y >= 0 && board[y][x]) return true;
      }
    }
    return false;
  }

  // ===== 操作 =====
  function move(dir) {
    if (paused || gameOver) return;
    if (!collide(current.shape, current.x + dir, current.y)) {
      current.x += dir;
    }
  }

  function rotate() {
    if (paused || gameOver) return;
    const rotated = rotateMatrix(current.shape);
    // 简单墙踢：尝试左右微调
    for (const dx of [0, -1, 1, -2, 2]) {
      if (!collide(rotated, current.x + dx, current.y)) {
        current.shape = rotated;
        current.x += dx;
        return;
      }
    }
  }

  function softDrop() {
    if (paused || gameOver) return;
    if (!collide(current.shape, current.x, current.y + 1)) {
      current.y++;
      score += 1;
      updateInfo();
    } else {
      lockPiece();
    }
    dropCounter = 0;
  }

  function hardDrop() {
    if (paused || gameOver) return;
    while (!collide(current.shape, current.x, current.y + 1)) {
      current.y++;
      score += 2;
    }
    updateInfo();
    lockPiece();
    dropCounter = 0;
  }

  // 固定方块到棋盘
  function lockPiece() {
    current.shape.forEach((row, r) => {
      row.forEach((val, c) => {
        if (val) {
          const y = current.y + r;
          const x = current.x + c;
          if (y >= 0) board[y][x] = current.color;
        }
      });
    });
    clearLines();
    spawnPiece();
  }

  // 消行
  function clearLines() {
    let cleared = 0;
    for (let r = ROWS - 1; r >= 0; r--) {
      if (board[r].every((cell) => cell)) {
        board.splice(r, 1);
        board.unshift(Array(COLS).fill(0));
        cleared++;
        r++; // 重新检查当前行
      }
    }
    if (cleared > 0) {
      // 计分：1行100，2行300，3行500，4行800，再乘等级
      const table = { 1: 100, 2: 300, 3: 500, 4: 800 };
      score += (table[cleared] || 0) * level;
      lines += cleared;
      level = Math.floor(lines / 10) + 1;
      dropInterval = Math.max(100, 800 - (level - 1) * 70);
      updateInfo();
    }
  }

  // 生成新方块
  function spawnPiece() {
    current = nextPiece || randomPiece();
    nextPiece = randomPiece();
    drawNext();

    // 出生即碰撞 => 游戏结束
    if (collide(current.shape, current.x, current.y)) {
      endGame();
    }
  }

  function endGame() {
    gameOver = true;
    overlayTitle.textContent = "游戏结束";
    overlayText.textContent = `最终分数：${score}`;
    overlay.classList.remove("hidden");
  }

  function updateInfo() {
    scoreEl.textContent = score;
    levelEl.textContent = level;
    linesEl.textContent = lines;
  }

  // ===== 绘制 =====
  function drawCell(context, x, y, color, size) {
    context.fillStyle = color;
    context.fillRect(x * size, y * size, size, size);
    context.strokeStyle = "rgba(0,0,0,0.35)";
    context.lineWidth = 2;
    context.strokeRect(x * size, y * size, size, size);
    // 高光
    context.fillStyle = "rgba(255,255,255,0.18)";
    context.fillRect(x * size + 3, y * size + 3, size - 6, 4);
  }

  function drawBoard() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // 网格背景
    ctx.strokeStyle = "rgba(255,255,255,0.05)";
    ctx.lineWidth = 1;
    for (let c = 1; c < COLS; c++) {
      ctx.beginPath();
      ctx.moveTo(c * BLOCK, 0);
      ctx.lineTo(c * BLOCK, canvas.height);
      ctx.stroke();
    }
    for (let r = 1; r < ROWS; r++) {
      ctx.beginPath();
      ctx.moveTo(0, r * BLOCK);
      ctx.lineTo(canvas.width, r * BLOCK);
      ctx.stroke();
    }

    // 已固定的方块
    for (let r = 0; r < ROWS; r++) {
      for (let c = 0; c < COLS; c++) {
        if (board[r][c]) drawCell(ctx, c, r, board[r][c], BLOCK);
      }
    }

    // 当前方块
    if (current) {
      current.shape.forEach((row, r) => {
        row.forEach((val, c) => {
          if (val) drawCell(ctx, current.x + c, current.y + r, current.color, BLOCK);
        });
      });
    }
  }

  function drawNext() {
    nctx.clearRect(0, 0, nextCanvas.width, nextCanvas.height);
    if (!nextPiece) return;
    const size = 24;
    const shape = nextPiece.shape;
    const w = shape[0].length * size;
    const h = shape.length * size;
    const ox = (nextCanvas.width - w) / 2;
    const oy = (nextCanvas.height - h) / 2;

    shape.forEach((row, r) => {
      row.forEach((val, c) => {
        if (!val) return;
        const x = ox + c * size;
        const y = oy + r * size;
        nctx.fillStyle = nextPiece.color;
        nctx.fillRect(x, y, size, size);
        nctx.strokeStyle = "rgba(0,0,0,0.35)";
        nctx.lineWidth = 2;
        nctx.strokeRect(x, y, size, size);
      });
    });
  }

  // ===== 主循环 =====
  function update(time = 0) {
    if (gameOver) return;
    const delta = time - lastTime;
    lastTime = time;

    if (!paused) {
      dropCounter += delta;
      if (dropCounter > dropInterval) {
        if (!collide(current.shape, current.x, current.y + 1)) {
          current.y++;
        } else {
          lockPiece();
        }
        dropCounter = 0;
      }
      drawBoard();
    }
    rafId = requestAnimationFrame(update);
  }

  // ===== 键盘控制 =====
  document.addEventListener("keydown", (e) => {
    switch (e.key) {
      case "ArrowLeft":
        move(-1);
        break;
      case "ArrowRight":
        move(1);
        break;
      case "ArrowDown":
        softDrop();
        break;
      case "ArrowUp":
        rotate();
        break;
      case " ":
        e.preventDefault();
        hardDrop();
        break;
      case "p":
      case "P":
        togglePause();
        break;
    }
  });

  function togglePause() {
    if (gameOver) return;
    paused = !paused;
    if (paused) {
      overlayTitle.textContent = "已暂停";
      overlayText.textContent = "按 P 继续游戏";
      overlay.classList.remove("hidden");
    } else {
      overlay.classList.add("hidden");
    }
  }

  // ===== 初始化 / 重开 =====
  function reset() {
    board = createBoard();
    score = 0;
    level = 1;
    lines = 0;
    dropInterval = 800;
    dropCounter = 0;
    lastTime = 0;
    paused = false;
    gameOver = false;
    nextPiece = null;
    current = null;

    overlay.classList.add("hidden");
    updateInfo();
    spawnPiece();
    drawBoard();

    if (rafId) cancelAnimationFrame(rafId);
    rafId = requestAnimationFrame(update);
  }

  restartBtn.addEventListener("click", reset);
  overlayBtn.addEventListener("click", reset);

  reset();
})();
```
