// 核心逻辑自测（node test-logic.mjs）
import {
  createBoard, clearLines, calcScore, spawnPiece, collides,
  merge, rotateMatrix, ghostY, ROWS, COLS, levelSpeed,
} from './src/tetris.js'

let pass = 0, fail = 0
const ok = (name, cond, extra = '') => {
  if (cond) { pass++; console.log('  PASS', name, extra) }
  else { fail++; console.log('  FAIL', name, extra) }
}

// 1. 消行
const b = createBoard()
b[19] = Array(COLS).fill('I')
b[18] = Array(COLS).fill('T')
b[17][0] = 'O'
const r = clearLines(b)
ok('消除 2 行', r.cleared === 2, `cleared=${r.cleared}`)
ok('棋盘行数不变', r.board.length === ROWS)
ok('顶部补空行', r.board[0].every((c) => !c))
ok('残留方块下移', r.board[19][0] === 'O')

// 2. 计分
ok('1行 L1 = 100', calcScore(1, 1, 0) === 100)
ok('4行 L1 = 800', calcScore(4, 1, 0) === 800)
ok('4行 L3 combo2 = 2500', calcScore(4, 3, 2) === 800 * 3 + 100, `got ${calcScore(4, 3, 2)}`)
ok('未消行 = 0', calcScore(0, 5, 3) === 0)

// 3. 生成 / 碰撞
const p = spawnPiece('T')
ok('T 出生居中', p.x === 3 && p.y === 0, `x=${p.x} y=${p.y}`)
ok('空棋盘不碰撞', collides(createBoard(), p) === false)
ok('左越界碰撞', collides(createBoard(), { ...p, x: -2 }) === true)
ok('右越界碰撞', collides(createBoard(), { ...p, x: COLS }) === true)
ok('底部越界碰撞', collides(createBoard(), { ...p, y: ROWS }) === true)

// 4. 合并
const merged = merge(createBoard(), { ...p, y: 18 })
ok('合并 4 格', merged.flat().filter(Boolean).length === 4)

// 5. 旋转
const rot = rotateMatrix([[1, 0, 0], [1, 1, 1], [0, 0, 0]])
ok('顺时针旋转', JSON.stringify(rot) === JSON.stringify([[0, 1, 1], [0, 1, 0], [0, 1, 0]]), JSON.stringify(rot))

// 6. 幽灵落点
ok('幽灵落到底部', ghostY(createBoard(), p) === ROWS - 2, `gy=${ghostY(createBoard(), p)}`)

// 7. 速度
ok('L1 速度 800ms', levelSpeed(1) === 800)
ok('速度有下限 80ms', levelSpeed(99) === 80)

console.log(`\n结果: ${pass} 通过, ${fail} 失败`)
process.exit(fail ? 1 : 0)

# 俄罗斯方块 · React Tetris

基于 React 18 + Vite 的俄罗斯方块小游戏，支持键盘方向键操作与实时计分。

## 运行

    cd D:\tetris-react
    npm install
    npm run dev      # 开发模式，自动打开 http://localhost:5173
    npm run build    # 打包到 dist/
    npm run preview  # 预览打包结果

## 操作

- 左 / 右：左右移动
- 上：旋转（带踢墙）
- 下：软降，每格 +1 分
- 空格：硬降到底，每格 +2 分
- P：暂停 / 继续
- Enter：开始 / 重新开始

移动端提供屏幕触控按钮。

## 计分规则

- 消除 1 行：100 分
- 消除 2 行：300 分
- 消除 3 行：500 分
- 消除 4 行（Tetris）：800 分
- 实际得分 = 基础分 x 当前等级
- 连续消行（Combo）额外 +50 x 连击数
- 软降 +1 分/格，硬降 +2 分/格
- 每消除 10 行升 1 级，下落速度加快（最快 80ms/格）
- 最高分保存在浏览器 localStorage

## 目录结构

    D:\tetris-react
    |- index.html
    |- package.json
    |- vite.config.js
    |- README.md
    |- src
       |- main.jsx      # 入口

       |- tetris.js     # 核心逻辑（方块、碰撞、消行、计分）
