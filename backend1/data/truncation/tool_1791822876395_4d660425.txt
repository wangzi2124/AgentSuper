278: 277: 276:  * 使用矩阵表示形状，旋转即矩阵转置 + 反转
279: 278: 277:  */
280: 279: 278: const SHAPES = {
281: 280: 279:   I: [[1, 1, 1, 1]],
282: 281: 280:   O: [
283: 282: 281:     [1, 1],
284: 283: 282:     [1, 1],
285: 284: 283:   ],
286: 285: 284:   T: [
287: 286: 285:     [0, 1, 0],
288: 287: 286:     [1, 1, 1],
289: 288: 287:   ],
290: 289: 288:   S: [
291: 290: 289:     [0, 1, 1],
292: 291: 290:     [1, 1, 0],
293: 292: 291:   ],
294: 293: 292:   Z: [
295: 294: 293:     [1, 1, 0],
296: 295: 294:     [0, 1, 1],
297: 296: 295:   ],
298: 297: 296:   J: [
299: 298: 297:     [1, 0, 0],
300: 299: 298:     [1, 1, 1],
301: 300: 299:   ],
302: 301: 300:   L: [
303: 302: 301:     [0, 0, 1],
304: 303: 302:     [1, 1, 1],
305: 304: 303:   ],
306: 305: 304: };
307: 306: 305: 
308: 307: 306: /** 每种方块对应的颜色 */
309: 308: 307: const COLORS = {
310: 309: 308:   I: "#4fd1c5",
311: 310: 309:   O: "#f6c445",
312: 311: 310:   T: "#b06cf0",
313: 312: 311:   S: "#5ad469",
314: 313: 312:   Z: "#ef5b6b",

318: 317: 316:

320: 319: 318:   /**
321: 320: 319:    * @param {string} type 方块类型（I/O/T/S/Z/J/L）
322: 321: 320:    */
323: 322: 321:   constructor(type) {
324: 323: 322:     this.type = type;
325: 324: 323:     // 深拷贝形状，避免旋转时污染原始定义
326: 325: 324:     this.matrix = SHAPES[type].map((row) => [...row]);
327: 326: 325:     this.color = COLORS[type];
328: 327: 326:     // 初始位置：水平居中，顶部
329: 328: 327:     this.x = Math.floor((CONFIG.COLS - this.matrix[0].length) / 2);
330: 329: 328:     this.y = 0;

335: 334: 333:     const rows = this.matrix.length;
336: 335: 334:     const cols = this.matrix[0].length;
337: 336: 335:     const result = Array.from({ length: cols }, () => Array(rows).fill(0));
338: 337: 336:     for (let r = 0; r < rows; r++) {
339: 338: 337:       for (let c = 0; c < cols; c++) {
340: 339: 338:         result[c][rows - 1 - r] = this.matrix[r][c];
341: 340: 339:       }
342: 341: 340:     }
343: 342: 341:     return result;
344: 343: 342:   }
345: 344: 343: 
346: 345: 344:   /** 执行旋转 */

355: 354: 353:       row.forEach((val, c) => {
356: 355: 354:         if (val) list.push({ x: offsetX + c, y: offsetY + r });
357: 356: 355:       });
358: 357: 356:     });
359: 358: 357:     return list;
360: 359: 358:   }

373: 372: 371: /**
374: 373: 372:  * 棋盘：负责网格数据、碰撞检测、锁定方块、消行
375: 374: 373:  */
376: 375: 374: class Board {
377: 376: 375:   constructor() {
378: 377: 376:     this.reset();
379: 378: 377:   }
380: 379: 378: 
381: 380: 379:   reset() {
382: 381: 380:     // 二维数组，null 表示空格，否则存储颜色
383: 382: 381:     this.grid = Array.from({ length: CONFIG.ROWS }, () =>
384: 383: 382:       Array(CONFIG.COLS).fill(null)
385: 384: 383:     );
386: 385: 384:   }
387: 386: 385: 
388: 387: 386:   /** 判断一组坐标是否越界或与已锁定方块重叠 */
389: 388: 387:   isCollision(cells) {
390: 389: 388:     return cells.some(({ x, y }) => {
391: 390: 389:       if (x < 0 || x >= CONFIG.COLS || y >= CONFIG.ROWS) return true;
392: 391: 390:       if (y < 0) return false; // 允许在顶部之上（生成时）
393: 392: 391:       return this.grid[y][x] !== null;
394: 393: 392:     });
395: 394: 393:   }
396: 395: 394: 
397: 396: 395:   /** 将方块锁定进网格 */
398: 397: 396:   lock(tetromino) {
399: 398: 397:     tetromino.cells().forEach(({ x, y }) => {
400: 399: 398:       if (y >= 0 && y < CONFIG.ROWS && x >= 0 && x < CONFIG.COLS) {
401: 400: 399:         this.grid[y][x] = tetromino.color;
402: 401: 400:       }
403: 402: 401:     });
404: 403: 402:   }
405: 404: 403: 
406: 405: 404:   /**
407: 406: 405:    * 消除已填满的行
408: 407: 406:    * @returns {number} 消除的行数
409: 408: 407:    */
410: 409: 408:   clearLines() {
411: 410: 409:     let cleared = 0;
412: 411: 410:     for (let r = CONFIG.ROWS - 1; r >= 0; r--) {
413: 412: 411:       if (this.grid[r].every((cell) => cell !== null)) {
414: 413: 412:         this.grid.splice(r, 1);
415: 414: 413:         this.grid.unshift(Array(CONFIG.COLS).fill(null));
416: 415: 414:         cleared++;
417: 416: 415:         r++; // 重新检查当前行（因为上方行下移了）
418: 417: 416:       }
419: 418: 417:     }
420: 419: 418:     return cleared;
421: 420: 419:   }
422: 421: 420: 
423: 422: 421:   /** 绘制棋盘与已锁定方块 */
424: 423: 422:   draw(ctx) {
425: 424: 423:     const { BLOCK_SIZE, COLS, ROWS } = CONFIG;
426: 425: 424:     ctx.clearRect(0, 0, COLS * BLOCK_SIZE, ROWS * BLOCK_SIZE);
427: 426: 425: 
428: 427: 426:     // 网格线
429: 428: 427:     ctx.strokeStyle = "rgba(255,255,255,0.05)";
430: 429: 428:     ctx.lineWidth = 1;
431: 430: 429:     for (let c = 0; c <= COLS; c++) {
432: 431: 430:       ctx.beginPath();
433: 432: 431:       ctx.moveTo(c * BLOCK_SIZE, 0);
434: 433: 432:       ctx.lineTo(c * BLOCK_SIZE, ROWS * BLOCK_SIZE);
435: 434: 433:       ctx.stroke();
436: 435: 434:     }
437: 436: 435:     for (let r = 0; r <= ROWS; r++) {
438: 437: 436:       ctx.beginPath();
439: 438: 437:       ctx.moveTo(0, r * BLOCK_SIZE);
440: 439: 438:       ctx.lineTo(COLS * BLOCK_SIZE, r * BLOCK_SIZE);
441: 440: 439:       ctx.stroke();
442: 441: 440:     }
443: 442: 441: 
444: 443: 442:     // 已锁定方块
445: 444: 443:     this.grid.forEach((row, r) => {
446: 445: 444:       row.forEach((color, c) => {
447: 446: 445:         if (color) drawBlock(ctx, c, r, color);
448: 447: 446:       });
449: 448: 447:     });
450: 449: 448:   }
451: 450: 449: }
452: 451: 450: 
453: 452: 451: /** 通用方块绘制函数（带高光效果） */
454: 453: 452: function drawBlock(ctx, col, row, color) {
455: 454: 453:   const { BLOCK_SIZE } = CONFIG;
456: 455: 454:   const x = col * BLOCK_SIZE;
457: 456: 455:   const y = row * BLOCK_SIZE;
458: 457: 456:   const pad = 1;
459: 458: 457: 
460: 459: 458:   ctx.fillStyle = color;
461: 460: 459:   ctx.fillRect(x + pad, y + pad, BLOCK_SIZE - pad * 2, BLOCK_SIZE - pad * 2);
462: 461: 460: 
463: 462: 461:   // 顶部/左侧高光
464: 463: 462:   ctx.fillStyle = "rgba(255,255,255,0.25)";
465: 464: 463:   ctx.fillRect(x + pad, y + pad, BLOCK_SIZE - pad * 2, 4);
