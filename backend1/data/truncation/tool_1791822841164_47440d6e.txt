274: 273: ```js
275: 274: /**
276: 275:  * 七种标准方块（Tetromino）定义与旋转逻辑
277: 276:  * 使用矩阵表示形状，旋转即矩阵转置 + 反转
278: 277:  */
279: 278: const SHAPES = {
280: 279:   I: [[1, 1, 1, 1]],
281: 280:   O: [
282: 281:     [1, 1],
283: 282:     [1, 1],
284: 283:   ],
285: 284:   T: [
286: 285:     [0, 1, 0],
287: 286:     [1, 1, 1],
288: 287:   ],
289: 288:   S: [
290: 289:     [0, 1, 1],
291: 290:     [1, 1, 0],
292: 291:   ],
293: 292:   Z: [
294: 293:     [1, 1, 0],
295: 294:     [0, 1, 1],
296: 295:   ],
297: 296:   J: [
298: 297:     [1, 0, 0],
299: 298:     [1, 1, 1],
300: 299:   ],
301: 300:   L: [
302: 301:     [0, 0, 1],
303: 302:     [1, 1, 1],
304: 303:   ],
305: 304: };
306: 305: 
307: 306: /** 每种方块对应的颜色 */
308: 307: const COLORS = {
309: 308:   I: "#4fd1c5",
310: 309:   O: "#f6c445",
311: 310:   T: "#b06cf0",
312: 311:   S: "#5ad469",
313: 312:   Z: "#ef5b6b",
314: 313:   J: "#4f8ef7",
315: 314:   L: "#f0913f",
316: 315: };
317: 316: 
318: 317: class Tetromino {
319: 318:   /**
320: 319:    * @param {string} type 方块类型（I/O/T/S/Z/J/L）
321: 320:    */
322: 321:   constructor(type) {
323: 322:     this.type = type;
324: 323:     // 深拷贝形状，避免旋转时污染原始定义
325: 324:     this.matrix = SHAPES[type].map((row) => [...row]);
326: 325:     this.color = COLORS[type];
327: 326:     // 初始位置：水平居中，顶部
328: 327:     this.x = Math.floor((CONFIG.COLS - this.matrix[0].length) / 2);
329: 328:     this.y = 0;
330: 329:   }
331: 330: 
332: 331:   /** 顺时针旋转 90°，返回新矩阵（不修改自身） */
333: 332:   rotatedMatrix() {
334: 333:     const rows = this.matrix.length;
335: 334:     const cols = this.matrix[0].length;
336: 335:     const result = Array.from({ length: cols }, () => Array(rows).fill(0));
337: 336:     for (let r = 0; r < rows; r++) {
338: 337:       for (let c = 0; c < cols; c++) {
339: 338:         result[c][rows - 1 - r] = this.matrix[r][c];
340: 339:       }
341: 340:     }
342: 341:     return result;
343: 342:   }
344: 343: 
345: 344:   /** 执行旋转 */
346: 345:   rotate() {
347: 346:     this.matrix = this.rotatedMatrix();
348: 347:   }
349: 348: 
350: 349:   /** 返回方块占据的绝对坐标列表 [{x, y}] */
351: 350:   cells(matrix = this.matrix, offsetX = this.x, offsetY = this.y) {
352: 351:     const list = [];
353: 352:     matrix.forEach((row, r) => {
354: 353:       row.forEach((val, c) => {
355: 354:         if (val) list.push({ x: offsetX + c, y: offsetY + r });
356: 355:       });
357: 356:     });
358: 357:     return list;
359: 358:   }
360: 359: }
361: 360: 
362: 361: /** 随机生成一个方块类型 */
363: 362: function randomType() {
364: 363:   const types = Object.keys(SHAPES);
365: 364:   return types[Math.floor(Math.random() * types.length)];
366: 365: }

372: 371: /**
373: 372:  * 棋盘：负责网格数据、碰撞检测、锁定方块、消行
374: 373:  */
375: 374: class Board {
376: 375:   constructor() {
377: 376:     this.reset();
378: 377:   }
379: 378: 
380: 379:   reset() {
381: 380:     // 二维数组，null 表示空格，否则存储颜色
382: 381:     this.grid = Array.from({ length: CONFIG.ROWS }, () =>
383: 382:       Array(CONFIG.COLS).fill(null)
384: 383:     );
385: 384:   }
386: 385: 
387: 386:   /** 判断一组坐标是否越界或与已锁定方块重叠 */
388: 387:   isCollision(cells) {
389: 388:     return cells.some(({ x, y }) => {
390: 389:       if (x < 0 || x >= CONFIG.COLS || y >= CONFIG.ROWS) return true;
391: 390:       if (y < 0) return false; // 允许在顶部之上（生成时）
392: 391:       return this.grid[y][x] !== null;
393: 392:     });
394: 393:   }
395: 394: 
396: 395:   /** 将方块锁定进网格 */
397: 396:   lock(tetromino) {
398: 397:     tetromino.cells().forEach(({ x, y }) => {
399: 398:       if (y >= 0 && y < CONFIG.ROWS && x >= 0 && x < CONFIG.COLS) {
400: 399:         this.grid[y][x] = tetromino.color;
401: 400:       }
402: 401:     });
403: 402:   }
404: 403: 
405: 404:   /**
406: 405:    * 消除已填满的行
407: 406:    * @returns {number} 消除的行数
408: 407:    */
409: 408:   clearLines() {
410: 409:     let cleared = 0;
411: 410:     for (let r = CONFIG.ROWS - 1; r >= 0; r--) {
412: 411:       if (this.grid[r].every((cell) => cell !== null)) {
413: 412:         this.grid.splice(r, 1);
414: 413:         this.grid.unshift(Array(CONFIG.COLS).fill(null));
415: 414:         cleared++;
416: 415:         r++; // 重新检查当前行（因为上方行下移了）
417: 416:       }
418: 417:     }
419: 418:     return cleared;
420: 419:   }
421: 420: 
422: 421:   /** 绘制棋盘与已锁定方块 */
423: 422:   draw(ctx) {
424: 423:     const { BLOCK_SIZE, COLS, ROWS } = CONFIG;
425: 424:     ctx.clearRect(0, 0, COLS * BLOCK_SIZE, ROWS * BLOCK_SIZE);
426: 425: 
427: 426:     // 网格线
428: 427:     ctx.strokeStyle = "rgba(255,255,255,0.05)";
429: 428:     ctx.lineWidth = 1;
430: 429:     for (let c = 0; c <= COLS; c++) {
431: 430:       ctx.beginPath();
432: 431:       ctx.moveTo(c * BLOCK_SIZE, 0);
433: 432:       ctx.lineTo(c * BLOCK_SIZE, ROWS * BLOCK_SIZE);
434: 433:       ctx.stroke();
435: 434:     }
436: 435:     for (let r = 0; r <= ROWS; r++) {
437: 436:       ctx.beginPath();
438: 437:       ctx.moveTo(0, r * BLOCK_SIZE);
439: 438:       ctx.lineTo(COLS * BLOCK_SIZE, r * BLOCK_SIZE);
440: 439:       ctx.stroke();
441: 440:     }
442: 441: 
443: 442:     // 已锁定方块
444: 443:     this.grid.forEach((row, r) => {
445: 444:       row.forEach((color, c) => {
446: 445:         if (color) drawBlock(ctx, c, r, color);
447: 446:       });
448: 447:     });
449: 448:   }
450: 449: }
451: 450: 
452: 451: /** 通用方块绘制函数（带高光效果） */
453: 452: function drawBlock(ctx, col, row, color) {
454: 453:   const { BLOCK_SIZE } = CONFIG;
455: 454:   const x = col * BLOCK_SIZE;
456: 455:   const y = row * BLOCK_SIZE;
457: 456:   const pad = 1;
458: 457: 
459: 458:   ctx.fillStyle = color;
460: 459:   ctx.fillRect(x + pad, y + pad, BLOCK_SIZE - pad * 2, BLOCK_SIZE - pad * 2);
461: 460: 
462: 461:   // 顶部/左侧高光
463: 462:   ctx.fillStyle = "rgba(255,255,255,0.25)";
464: 463:   ctx.fillRect(x + pad, y + pad, BLOCK_SIZE - pad * 2, 4);
465: 464:   ctx.fillRect(x + pad, y + pad, 4, BLOCK_SIZE - pad * 2);
